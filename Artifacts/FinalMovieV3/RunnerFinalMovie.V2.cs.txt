using System;
using System.Collections.Generic;
using UnityEngine;
namespace SuyuRun
{
    // Authored post-game film. DSP clock is the sole timeline; no gameplay or save mutations.
    public sealed class RunnerFinalMovie:MonoBehaviour
    {
        public static bool Active {get;private set;}
        public const float FilmSeconds=20,CreditsSeconds=10,TotalSeconds=30;
        double began;AudioSource track;float volume;
        readonly Dictionary<string,Texture2D> textures=new Dictionary<string,Texture2D>();
        readonly Dictionary<string,Sprite[]> actors=new Dictionary<string,Sprite[]>();
        [Serializable] sealed class Spec {public int frames,frameWidth,frameHeight,fps;public bool loop;}
        readonly Dictionary<string,Spec> specs=new Dictionary<string,Spec>();
        public static void Begin(GameObject host){if(!Active)host.AddComponent<RunnerFinalMovie>();}
        void Awake()
        {
            Active=true;began=AudioSettings.dspTime;
            volume=.34f*PlayerPrefs.GetFloat("SuyuRun.Settings.MusicVol",1);
            track=gameObject.AddComponent<AudioSource>();track.clip=Resources.Load<AudioClip>("FinalMovie/final_music");
            track.loop=true;track.playOnAwake=false;track.spatialBlend=0;track.volume=0;
            if(track.clip)track.Play();
        }
        void Update()
        {
            float age=(float)(AudioSettings.dspTime-began);
            if(track)track.volume=volume*Mathf.SmoothStep(0,1,age/.65f)*Mathf.SmoothStep(0,1,(TotalSeconds-age)/1.5f);
            if(age>=TotalSeconds)Destroy(this);
        }
        void OnDestroy(){Active=false;if(track){track.Stop();Destroy(track);}foreach(var sheet in actors.Values)foreach(var s in sheet)if(s)Destroy(s);}
        Texture2D Texture(string resource)
        {
            if(!textures.TryGetValue(resource,out var t))
            {t=RefactorMaterialArt.FullTexture(Resources.Load<Texture2D>(resource));if(t){t.filterMode=FilterMode.Point;t.wrapMode=TextureWrapMode.Clamp;}textures[resource]=t;}
            return t;
        }
        void Layer(string resource,float time,float speed,float top=0,float height=180)
        {
            var t=Texture(resource);if(!t)return;
            float x=-Mathf.Repeat(time*speed,640);
            GUI.DrawTexture(new Rect(x,top,640,height),t,ScaleMode.StretchToFill,true);
            if(speed!=0)GUI.DrawTexture(new Rect(x+640,top,640,height),t,ScaleMode.StretchToFill,true);
        }
        void World(int region,float time,float pace=1)
        {
            if(region==0)
            {
                Layer("costa_capa1_cielo",0,0);Layer("costa_capa2_nubes",time,2*pace);
                Layer("costa_capa3_paracas",time,8*pace);Layer("costa_capa5_huacas",time,22*pace);
                // The coastal ground is a tile, not a full-height backdrop.
                var ground=Texture("costa_capa7_suelo");
                if(ground)for(int i=-1;i<12;i++)GUI.DrawTexture(new Rect(i*32-Mathf.Repeat(time*60*pace,32),146,32,32),ground);
            }
            else if(region==1)
            {
                Layer("SierraV3/sierra_compartido_capa1_cielo",0,0);
                Layer("SierraV3/sierra_compartido_capa3_nevados",time,7*pace);
                Layer("SierraV3/sierra_camino_capa4_paisaje",time,18*pace);
                Layer("SierraV3/sierra_camino_capa7_suelo",time,55*pace);
            }
            else
            {
                Layer("SelvaV2/selva_bosque_capa1_cielo",0,0);
                Layer("SelvaV2/selva_bosque_capa2_lejania",time,7*pace);
                Layer("SelvaV2/selva_bosque_capa3_estructura",time,20*pace);
                Layer("SelvaV2/selva_bosque_capa5_cercania",time,38*pace);
                var tile=Texture("SelvaV2/selva_tile_ruina_superficie");
                if(tile)for(int i=-1;i<12;i++)GUI.DrawTexture(new Rect(i*32-Mathf.Repeat(time*60*pace,32),146,32,32),tile);
            }
        }
        void Actor(string path,Rect r,float age,int forced=-1)
        {
            if(!actors.TryGetValue(path,out var frames))
            {
                var t=Resources.Load<Texture2D>(path);var data=Resources.Load<TextAsset>(path);if(!t||!data)return;
                var s=JsonUtility.FromJson<Spec>(data.text);
                if(s.frames<1||t.width!=s.frames*s.frameWidth||t.height!=s.frameHeight)return;
                t.filterMode=FilterMode.Point;frames=new Sprite[s.frames];
                for(int i=0;i<s.frames;i++)frames[i]=Sprite.Create(t,new Rect(i*s.frameWidth,0,s.frameWidth,s.frameHeight),new Vector2(.5f,0),128);
                actors[path]=frames;specs[path]=s;
            }
            var spec=specs[path];int f=forced>=0?forced:Mathf.FloorToInt(Mathf.Max(age,0)*Mathf.Max(1,spec.fps));
            RunnerHud.Sprite(r,frames[spec.loop?f%frames.Length:Mathf.Clamp(f,0,frames.Length-1)]);
        }
        void Hero(string region,string state,float age,float x=130,float feet=146,float scale=1.2f,int forced=-1)
        {
            var sprite=TaytaArt.Frame(region,state,age,forced)??TaytaArt.Frame(region,"idle",age);
            if(!sprite)return;var b=sprite.bounds;
            RunnerHud.Sprite(new Rect(x+b.min.x*32*scale,feet-47*scale-b.max.y*32*scale,b.size.x*32*scale,b.size.y*32*scale),sprite);
        }
        void Title(string text,float y,int size,Color color)
        {
            var s=RunnerHud.Centered(new GUIStyle(GUI.skin.label){fontSize=size});
            s.normal.textColor=Color.black;GUI.Label(new Rect(13,y+1,296,18),text,s);
            s.normal.textColor=color;GUI.Label(new Rect(12,y,296,18),text,s);
        }
        void Dust(float t,float x,float feet)
        {
            if(t<0||t>.6f)return;
            Actor("RefactorBank/Materials/selva_fx_polvo",new Rect(x-18,feet-18,36,24),t);
        }
        void Coast(float t)
        {
            World(0,t);
            if(t<.8f)Hero("costa","run",t);
            else if(t<2.5f)
            {
                float u=(t-.8f)/1.7f,lift=4*42*u*(1-u);
                Hero("costa",u<.5f?"jump":"fall",u<.5f?t-.8f:t-1.65f,130,146-lift);
            }
            else if(t<2.75f){Hero("costa","land",t-2.5f);Dust(t-2.5f,130,146);}
            else Hero("costa","slide",t-2.75f,130,146);
            Actor("RefactorBank/Actors/coastbird/flight",new Rect(260-t*25,45,44,44),t);
        }
        void Sierra(float t)
        {
            World(1,t);
            string state=t<.5f?"run":t<1?"huaraca_cargar":t<2.1f?"huaraca_girar":t<2.6f?"huaraca_lanzar":"run";
            float pose=t<(float).5?t:t<1?t-.5f:t<2.1f?t-1:t<2.6f?t-2.1f:t-2.6f;
            Hero("sierra",state,pose);
            Actor("RefactorBank/Actors/condor/flight",new Rect(255-t*22,38,56,56),t);
            if(t>=2.35f&&t<3.2f)
                Actor("RefactorBank/Readability/sierra_piedra",new Rect(168+(t-2.35f)*140,92-(t-2.35f)*18,6,6),t-2.35f);
        }
        void Selva(float t)
        {
            World(2,t);
            if(t<2.1f)
            {
                float u=t/2.1f,angle=Mathf.Lerp(-.7f,.65f,Mathf.SmoothStep(0,1,u));
                Vector2 pivot=new Vector2(156,0),hand=pivot+new Vector2(Mathf.Sin(angle)*92,Mathf.Cos(angle)*92);
                // Curved rope with a single hand anchor, not a rigid rotating pole.
                Vector2 last=pivot;
                for(int i=1;i<=18;i++)
                {
                    float q=i/18f;Vector2 next=Vector2.Lerp(pivot,hand,q)+new Vector2(Mathf.Sin(q*Mathf.PI)*Mathf.Sin(angle)*5,0);
                    Vector2 delta=next-last;var m=GUI.matrix;
                    GUIUtility.RotateAroundPivot(Mathf.Atan2(delta.y,delta.x)*Mathf.Rad2Deg,last);
                    RunnerPresentation.Box(new Rect(last.x,last.y,delta.magnitude+1,1),new Color32(143,107,66,255));GUI.matrix=m;last=next;
                }
                int pose=TaytaArt.SwingPose(angle*Mathf.Rad2Deg);
                Vector2 grip=TaytaArt.GripOffset("swing",pose);
                Hero("selva","swing",t,hand.x-grip.x*1.2f,hand.y+(47-grip.y)*1.2f,1.2f,pose);
            }
            else if(t<2.85f)
            {
                float u=(t-2.1f)/.75f;Hero("selva","fall",t-2.1f,Mathf.Lerp(190,144,u),Mathf.Lerp(120,146,u*u));
            }
            else if(t<3.08f){Hero("selva","land",t-2.85f,144);Dust(t-2.85f,144,146);}
            else
            {
                Hero("selva","jump_shoot",t-3.08f,144,146-18*Mathf.Sin(Mathf.Clamp01((t-3.08f)/.92f)*Mathf.PI));
                if(t>3.35f&&t<3.65f)RunnerPresentation.Box(new Rect(182+(t-3.35f)*280,111,5,1),RunnerPresentation.Gold);
            }
        }
        void Portrait(float time,bool farewell)
        {
            World(2,time,.12f);
            float phase=farewell?Mathf.Clamp01(time/1.8f):Mathf.Clamp01(time/2.3f);
            int frame=farewell?3+Mathf.Min(2,Mathf.FloorToInt(phase*3)):Mathf.Min(2,Mathf.FloorToInt(phase*3));
            Actor("FinalMovieV2/tayta_portrait",new Rect(90,30+Mathf.Sin(time*3)*.45f,140,140),time,frame);
            if(farewell){Title("SUYURUN",120,11,RunnerPresentation.Gold);Title("EL VIAJE CONTINUA",141,4,Color.white);}
        }
        void Memories(float time)
        {
            int slot=0;
            for(int region=0;region<3;region++)
            {
                int count=region==1?3:4;
                for(int i=0;i<count;i++)
                {
                    if(!SharedGameMenu.IsCollected(region,i))continue;
                    string id=region==0?CoastalCulture.Entries[i].id:region==1?SharedGameMenu.SierraIds[i]:SharedGameMenu.SelvaIds[i];
                    float enter=Mathf.Clamp01((time-slot*.06f)/.3f);
                    if(enter<=0)continue;
                    var old=GUI.color;GUI.color=new Color(1,1,1,enter);
                    MuseumArt.Draw(new Rect(18+slot*26,24+(1-enter)*5,23,23),id,true);GUI.color=old;slot++;
                }
            }
        }
        void Credits(float time)
        {
            string[] lines={"SUYURUN","CREADO POR","ROMMEL C. V.","WILLIAM R. Q.","JOE F. C.","PROFESOR","MG. SULLA TORRES","DESARROLLO DE SOFTWARE","PARA JUEGOS","UNSA","COSTA · SIERRA · SELVA","GRACIAS POR JUGAR"};
            // Entire roll travels through the safe region before the final hold.
            float y=164-time*48;
            for(int i=0;i<lines.Length;i++)Title(lines[i],y+i*19,i==0?9:4,i==0?RunnerPresentation.Gold:Color.white);
            RunnerPresentation.Box(new Rect(0,0,320,12),Color.black);
            RunnerPresentation.Box(new Rect(0,168,320,12),Color.black);
            if(time>8.5f)Title("GRACIAS POR JUGAR",81,6,RunnerPresentation.Gold);
        }
        void OnGUI()
        {
            var matrix=GUI.matrix;var color=GUI.color;int depth=GUI.depth;
            try
            {
                GUI.depth=-1000;GUI.color=Color.white;
                // Cover letterbox areas too; the gameplay must never show behind the film.
                GUI.matrix=Matrix4x4.identity;RunnerPresentation.Box(new Rect(0,0,Screen.width,Screen.height),Color.black);
                GUI.matrix=RunnerHud.ScreenMatrix;
                float age=Mathf.Max(0,(float)(AudioSettings.dspTime-began));
                if(age<3)Portrait(age,false);
                else if(age<7)Coast(age-3);
                else if(age<11)Sierra(age-7);
                else if(age<15)Selva(age-11);
                else if(age<17.5f){World(2,age-15,.25f);Hero("selva","victory",age-15,145);Memories(age-15);}
                else if(age<20)Portrait(age-17.5f,true);
                else Credits(age-20);
                if(age<20)
                {
                    RunnerPresentation.Box(new Rect(0,0,320,12),Color.black);
                    RunnerPresentation.Box(new Rect(0,168,320,12),Color.black);
                }
                // Match cuts at 7/11 retain motion. Only opening/closing and the
                // time-jump from the portrait dissolve through darkness.
                float fade=age<.5f?1-age/.5f:age>19.4f&&age<20?(age-19.4f)/.6f:0;
                if(age>=2.8f&&age<3.2f)fade=1-Mathf.Abs(age-3)/.2f;
                if(age>=17.3f&&age<17.7f)fade=1-Mathf.Abs(age-17.5f)/.2f;
                if(age>29.5f)fade=(age-29.5f)/.5f;
                if(fade>0)RunnerPresentation.Box(new Rect(0,0,320,180),new Color(0,0,0,Mathf.Clamp01(fade)));
            }
            finally{GUI.matrix=matrix;GUI.color=color;GUI.depth=depth;}
        }
    }
}
